IMPORT HOOKS:

    useSyncCategories, 
    useSyncServices, 
    useSyncUserOrders,
    useSyncUserFavorites 
  } from '@/lib/hooks/useDataSync'

USE IN COMPONENT:

  const { services } = useSyncServices(categoryId)
  const { orders } = useSyncUserOrders(userId)
  const { favorites } = useSyncUserFavorites(userId)

  import { createOrder, getUserOrders } from '@/lib/database/orders'

  const categories = await getCategories()

  await addFavorite(userId, serviceId)
